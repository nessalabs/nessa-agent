import os
import pathlib
import subprocess

binary = '/workspace/nessa-agent/target/debug/deps/nessa_sdk-438bd3a05b08368c'
test = 'infrastructure::process::tests::private_directory_is_removed_only_after_process_cleanup_is_confirmed'
before = {p.name for p in pathlib.Path('/proc').iterdir() if p.name.isdigit()}
args = [binary, test, '--exact', '--nocapture']
print('Unchanged isolated test:', args, flush=True)
result = subprocess.run(args, cwd='/workspace/nessa-agent', stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
print(result.stdout, end='')
print('actual test exit:', result.returncode)
for p in pathlib.Path('/proc').iterdir():
    if not p.name.isdigit() or p.name in before:
        continue
    try:
        raw = (p / 'stat').read_text()
        name = raw[raw.index('(')+1:raw.rindex(')')]
        fields = raw[raw.rindex(')')+2:].split()
        state, parent, group = fields[:3]
        if state == 'Z' and parent == '1':
            try:
                os.killpg(int(group), 0)
                probe = 'group still exists (signal 0 succeeded)'
            except ProcessLookupError:
                probe = 'ESRCH'
            print(f'new orphan: pid={p.name} ppid={parent} pgid={group} state={state} comm={name}; {probe}')
    except (FileNotFoundError, ProcessLookupError):
        pass
print('Counterpart: same binary, exact filter and flags, task-local subreaper only', flush=True)
other = subprocess.run(['python', '/tmp/nessa-test-subreaper.py', *args], cwd='/workspace/nessa-agent')
print('supervised command exit:', other.returncode)
