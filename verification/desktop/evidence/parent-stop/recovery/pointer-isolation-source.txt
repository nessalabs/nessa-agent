import assert from 'node:assert/strict'
import { readFileSync } from 'node:fs'
import { randomUUID } from 'node:crypto'
import { setTimeout as sleep } from 'node:timers/promises'
import vm from 'node:vm'
import { chromium } from '/workspace/nessa-981-current/node_modules/playwright/index.mjs'
const source = readFileSync('/workspace/nessa-981-current/verification/desktop/scripts/mcp-apps-gateway.mjs','utf8')
const from = source.indexOf('async function pointerOnto(')
const to = source.indexOf('/** Waits up to', from)
assert(from >= 0 && to > from)
const pointerOnto = vm.runInNewContext(`(${source.slice(from,to).trim()})`, {
  randomUUID, sleep, appReviewUnstartedMs:500,
  selectorFor:{reviewControl:name=>`[data-review="${name}"]`},
})
const browser = await chromium.launch({executablePath:'/usr/bin/chromium'})
try {
  const page = await browser.newPage()
  await page.setContent('<style>button{position:absolute;top:80px}#m{left:50px}#c{left:250px}</style><button id="m" data-review="message">Message</button><button id="c" data-review="context">Context</button>')
  const frame=page.mainFrame()
  const message=await pointerOnto(page,frame,'message')
  assert(message>0,'trusted Message input reaches Message')
  const wrongTarget={
    evaluate:frame.evaluate.bind(frame),
    locator:()=>({hover:()=>page.locator('#m').hover()}),
  }
  const contextFromMessage=await pointerOnto(page,wrongTarget,'context')
  assert.equal(contextFromMessage,0,'trusted Message events cannot certify Context entry')
  const context=await pointerOnto(page,frame,'context')
  assert(context>0,'trusted Context input reaches Context')
  console.log(JSON.stringify({message,contextFromMessage,context,actualBrowser:await browser.version(),ok:true}))
} finally {await browser.close()}
